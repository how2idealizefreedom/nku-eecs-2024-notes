        function processAndShowCrop(app, currentPosition)
            
            croppedImg = imcrop(app.ImgOriginal, currentPosition);
            
            if size(croppedImg, 3) == 3
                grayCrop = rgb2gray(croppedImg);
            else
                grayCrop = croppedImg;
            end
            
            edgeCrop = edge(grayCrop, 'Canny');
            
            imshow(edgeCrop, 'Parent', app.UIAxes_SegCrop);
            title(app.UIAxes_SegCrop, 'ROI实时智能轮廓提取');
            
        end
        
    end

        % Button pushed function: Btn_Load
        function Btn_LoadButtonPushed(app, event)
           
            [file, path] = uigetfile({'*.jpg;*.png;*.bmp;*.tif', '图像文件'});
            
            if isequal(file, 0)
                return;
            end
            
            fullFileName = fullfile(path, file);
            app.ImgOriginal = imread(fullFileName); 
            app.ImgCurrent = app.ImgOriginal;       
            
            imshow(app.ImgOriginal, 'Parent', app.UIAxes_Original);
            
            imshow(app.ImgCurrent, 'Parent', app.UIAxes_Result);
            
            uialert(app.UIFigure, '图像加载成功！请选择算法进行处理。', '系统提示', 'Icon', 'success');
        end

        % Value changed function: Drop_Algo
        function Drop_AlgoValueChanged(app, event)
            if isempty(app.ImgOriginal)
                uialert(app.UIFigure, '请先导入图像！', '操作错误', 'Icon', 'warning');
                return;
            end
            
            selectedAlgo = app.Drop_Algo.Value;
            
            d = uiprogressdlg(app.UIFigure, 'Title', '算法运行中', 'Message', ['正在执行: ', selectedAlgo]);
            
            try
                switch selectedAlgo
                    case '限制对比度增强'
                        clipLimitValue = 0.1; 
                        
                        if size(app.ImgOriginal, 3) == 3
                            labImg = rgb2lab(app.ImgOriginal);
                            L = labImg(:,:,1) / 100;
                            L_enhanced = adapthisteq(L, 'NumTiles', [4 4], 'ClipLimit', clipLimitValue);
                            labImg(:,:,1) = L_enhanced * 100;
                            app.ImgCurrent = lab2rgb(labImg);
                        else
                            app.ImgCurrent = adapthisteq(app.ImgOriginal, 'NumTiles', [4 4], 'ClipLimit', clipLimitValue);
                        end
                        
                    case '伪彩色映射'
                        if size(app.ImgOriginal, 3) == 3
                            grayImg = rgb2gray(app.ImgOriginal);
                        else
                            grayImg = app.ImgOriginal;
                        end
                        % 核心映射代码
                        app.ImgCurrent = ind2rgb(grayImg, jet(256));
                        
                    case '边缘提取'
                        if size(app.ImgOriginal, 3) == 3
                            grayImg = rgb2gray(app.ImgOriginal);
                        else
                            grayImg = app.ImgOriginal;
                        end
                        edgeImg = edge(grayImg, 'Canny');
                        app.ImgCurrent = uint8(edgeImg) * 255; 
                end
                
                imshow(app.ImgCurrent, 'Parent', app.UIAxes_Result);
                
            catch ME
                uialert(app.UIFigure, ['处理失败: ' ME.message], '算法错误');
            end
            
            close(d);value = app.Drop_Algo.Value;
            
        end

        function Btn_ResetButtonPushed(app, event)
            app.ImgCurrent = app.ImgOriginal; 
            imshow(app.ImgCurrent, 'Parent', app.UIAxes_Result); 
        end

        % Button pushed function: Btn_3D
        function Btn_3DButtonPushed(app, event)
            
            if isempty(app.ImgOriginal)
                uialert(app.UIFigure, '请先在第一页导入图像！', '提示');
                return;
            end
            
            
            d = uiprogressdlg(app.UIFigure, 'Title', '构建多维矩阵', 'Message', '正在进行3D渲染，请稍候...');
            
            if size(app.ImgOriginal, 3) == 3
                grayImg = rgb2gray(app.ImgOriginal);
            else
                grayImg = app.ImgOriginal;
            end
            smallImg = imresize(grayImg, [200, 200]); 
            
            surf(app.UIAxes_3D, double(smallImg), 'EdgeColor', 'none');
            
            colormap(app.UIAxes_3D, 'parula'); 
            view(app.UIAxes_3D, -37.5, 30);    
            grid(app.UIAxes_3D, 'on');     
            
            title(app.UIAxes_3D, '图像空间亮度3D拓扑结构');
            xlabel(app.UIAxes_3D, 'X轴像素坐标');
            ylabel(app.UIAxes_3D, 'Y轴像素坐标');
            zlabel(app.UIAxes_3D, '像素强度 (灰度值)');
            
            enableDefaultInteractivity(app.UIAxes_3D);
            
            close(d);
        end

        % Button pushed function: Btn_ROI
        function Btn_ROIButtonPushed(app, event)
            if isempty(app.ImgOriginal)
                uialert(app.UIFigure, '请先在第一页导入图像！', '操作错误');
                return;
            end
            
            disableDefaultInteractivity(app.UIAxes_SegMain);
            
            imshow(app.ImgOriginal, 'Parent', app.UIAxes_SegMain);
            
            try
                roi = drawrectangle(app.UIAxes_SegMain, 'Label', '目标提取区', 'Color', 'r', 'LineWidth', 2);
                
                app.processAndShowCrop(roi.Position);

                addlistener(roi, 'MovingROI', @(src, event) app.processAndShowCrop(event.CurrentPosition));
                
            catch 
                disp('用户取消了框选操作');
            end
        end

        % Value changed function: Slider_Intensity
        function Slider_IntensityValueChanged(app, event)
            value = app.Slider_Intensity.Value;
            
            if isempty(app.ImgOriginal)
                return;
            end
            
            sliderVal = app.Slider_Intensity.Value;
            
            selectedAlgo = app.Drop_Algo.Value;
            
            switch selectedAlgo
                case '限制对比度增强'
                    clipLimitValue = 0.01 + (sliderVal / 100) * 0.19; 
                    
                    if size(app.ImgOriginal, 3) == 3
                        labImg = rgb2lab(app.ImgOriginal);
                        L = labImg(:,:,1) / 100;
                        L_enhanced = adapthisteq(L, 'NumTiles', [4 4], 'ClipLimit', clipLimitValue);
                        labImg(:,:,1) = L_enhanced * 100;
                        app.ImgCurrent = lab2rgb(labImg);
                    else
                        app.ImgCurrent = adapthisteq(app.ImgOriginal, 'NumTiles', [4 4], 'ClipLimit', clipLimitValue);
                    end
                    
                case '边缘提取'
                    thresh = 0.1 + ((100 - sliderVal) / 100) * 0.7; 
                    
                    if size(app.ImgOriginal, 3) == 3
                        grayImg = rgb2gray(app.ImgOriginal);
                    else
                        grayImg = app.ImgOriginal;
                    end
                    edgeImg = edge(grayImg, 'Canny', thresh);
                    app.ImgCurrent = uint8(edgeImg) * 255;
                    
                case '伪彩色映射'
                    alpha = sliderVal / 100; 
                    
                    if size(app.ImgOriginal, 3) == 3
                        grayImg = rgb2gray(app.ImgOriginal);
                    else
                        grayImg = app.ImgOriginal;
                    end
                    
                    pseudoImg = ind2rgb(grayImg, jet(256));
                    grayRGB = cat(3, grayImg, grayImg, grayImg) / 255;
                    
                    app.ImgCurrent = alpha * pseudoImg + (1 - alpha) * double(grayRGB);
            end
            
            imshow(app.ImgCurrent, 'Parent', app.UIAxes_Result);
        end

        % Button pushed function: Btn_ExportData
        function Btn_ExportDataPushed(app, event)
            if isempty(app.ImgCurrent)
                uialert(app.UIFigure, '当前没有可导出的图像，请先导入并处理！', '操作错误', 'Icon', 'warning');
                return;
            end
            
            [file, path] = uiputfile({ ...
                '*.png', '无损PNG 图像 (*.png)'; ...
                '*.jpg', '高压JPEG 图像 (*.jpg)'; ...
                '*.bmp', '位图 BMP (*.bmp)'}, ...
                '保存处理结果', 'Processed_Image.png');
            
            if isequal(file, 0) || isequal(path, 0)
                return;
            end
            
            fullFileName = fullfile(path, file);
            
            try
             
                d = uiprogressdlg(app.UIFigure, 'Title', '导出中', 'Message', '正在写入磁盘...');
                
                imwrite(app.ImgCurrent, fullFileName);
                
                close(d);
               
                uialert(app.UIFigure, ['图像已成功保存至：', newline, fullFileName], '导出成功', 'Icon', 'success');
                
            catch ME
                close(d);
                uialert(app.UIFigure, ['保存失败，请检查路径权限。', newline, '错误信息: ', ME.message], '导出失败', 'Icon', 'error');
            end
        end
    end
